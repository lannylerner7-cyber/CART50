const $=id=>document.getElementById(id);let quote=null,order=null,checking=false,redirecting=false;
let draft;try{draft=JSON.parse(localStorage.getItem('mainvault-order')||'null');}catch{}
const cents=s=>/^\d+(\.\d{1,2})?$/.test(s)?Number(s.split('.')[0])*100+Number((s.split('.')[1]||'').padEnd(2,'0')):null;
async function request(path,data){const r=await fetch(path,{method:data?'POST':'GET',headers:{'Content-Type':'application/json'},...(data?{body:JSON.stringify(data)}:{})});const j=await r.json();if(!r.ok)throw Error(j.error||'Request failed');return j;}
function estimate(){const n=cents($('amount').value);$('estimate').textContent=quote&&Date.now()-Date.parse(quote.updatedAt)<=180000&&n!==null?'Approximately '+(n/100/quote.price).toFixed(8)+' simulated BTC · final amount set on creation':'USD reference unavailable until a fresh quote is received.';}
function show(o){order=o;$('order').hidden=false;$('form').hidden=true;$('reference').textContent=o.orderId;$('btc').textContent=(o.amount_sats/1e8).toFixed(8)+' simulated BTC';$('status').textContent=o.status;$('qr').src=o.qrUrl;$('hosted').href=o.checkoutUrl;$('new').hidden=!['simulation_paid','expired'].includes(o.status);if(o.status==='simulation_paid'&&!redirecting){redirecting=true;$('status').textContent='Simulation received. Returning…';setTimeout(()=>location.assign(o.returnUrl),2500);}}
async function check(){if(!order||checking)return;checking=true;$('check').disabled=true;try{show(await request('/api/orders/'+encodeURIComponent(order.orderId)));$('error').textContent='';}catch(e){$('error').textContent=e.message;}finally{checking=false;$('check').disabled=false;}}
$('form').onsubmit=async e=>{e.preventDefault();const n=cents($('amount').value);if(!Number.isSafeInteger(n)||n<1)return;$('create').disabled=true;try{if(!draft){draft={id:crypto.randomUUID(),cents:n};localStorage.setItem('mainvault-order',JSON.stringify(draft));}if(draft.cents!==n)throw Error('Retry the saved amount first. An earlier request may have created an order.');show(await request('/api/orders',draft));$('error').textContent='';}catch(e){$('error').textContent=e.message;}finally{$('create').disabled=false;}};
$('check').onclick=check;$('amount').oninput=estimate;$('new').onclick=()=>{localStorage.removeItem('mainvault-order');location.reload();};
request('/api/config').then(c=>{$('return').textContent='Return to '+c.pageName;$('return').href=c.returnUrl;}).catch(e=>$('error').textContent=e.message);
async function refreshQuote(){try{quote=await request('/api/quote');}catch{quote=null;}estimate();}
refreshQuote();setInterval(refreshQuote,60000);setInterval(check,10000);
if(draft){$('amount').value=(draft.cents/100).toFixed(2);request('/api/orders/'+encodeURIComponent(draft.id)).then(show).catch(()=>$('error').textContent='Saved order retained. Retry creation with this same amount to reconcile its status.');}
